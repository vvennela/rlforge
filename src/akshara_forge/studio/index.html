// Find a path within a growing cost bound.
function IDA_STAR(start, goal):
    bound ← heuristic(start)
    loop:
        result ← SEARCH([start], 0, bound)
        if result is FOUND: return result.path
        if result = ∞: return NO_PATH
        bound ← result

function SEARCH(path, cost, bound):
    node ← last(path)
    f ← cost + heuristic(node)
    if f > bound: return f
    if node = goal: return FOUND(path)
    nextBound ← ∞
    for child in successors(node):
        if child in path: continue
        result ← SEARCH(path + child,
                        cost + edgeCost(node, child), bound)
        if result is FOUND: return result
        nextBound ← min(nextBound, result)
    return nextBound
